    activeTab = name

    for tabName, button in pairs(tabButtons) do
        button.BackgroundColor3 = tabName == name and UI_PURPLE or Color3.fromRGB(39, 25, 54)
        button.TextColor3 = UI_TEXT
    end
end

local function pointer(input) return input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch end
Launcher.InputBegan:Connect(function(input)
    if pointer(input) then dragTarget = "Launcher"; dragStart = input.Position; dragPosition = Launcher.AbsolutePosition end
end)
Header.InputBegan:Connect(function(input)
    if pointer(input) and input.Position.X < Close.AbsolutePosition.X then dragTarget = "Main"; dragStart = input.Position; dragPosition = Main.AbsolutePosition end
end)
mainGrip.InputBegan:Connect(function(input) if pointer(input) then resizeTarget = "Main"; resizeStart = input.Position; resizeSize = MenuSize end end)
launcherGrip.InputBegan:Connect(function(input) if pointer(input) then resizeTarget = "Launcher"; resizeStart = input.Position; resizeSize = Launcher.AbsoluteSize end end)
UserInputService.InputChanged:Connect(function(input)
    if input.UserInputType ~= Enum.UserInputType.MouseMovement and input.UserInputType ~= Enum.UserInputType.Touch then return end
    if resizeTarget then
        local delta = input.Position - resizeStart
        if resizeTarget == "Main" then
            local w = math.clamp(resizeSize.X + delta.X / MainScale.Scale, 560, 1100)
            local h = math.clamp(resizeSize.Y + delta.Y / MainScale.Scale, 350, 750)
            MenuSize = Vector2.new(w,h)
            Main.Size = UDim2.fromOffset(w,h)
        else
            Launcher.Size = UDim2.fromOffset(math.clamp(resizeSize.X + delta.X,105,300), math.clamp(resizeSize.Y + delta.Y,34,85))
        end
    elseif dragTarget then
        local delta = input.Position - dragStart
        if dragTarget == "Launcher" then
            local camera = workspace.CurrentCamera
            if camera then
                local v = camera.ViewportSize
                Launcher.Position = UDim2.fromOffset(math.clamp(dragPosition.X+delta.X,0,v.X-Launcher.AbsoluteSize.X), math.clamp(dragPosition.Y+delta.Y,0,v.Y-Launcher.AbsoluteSize.Y))
            end
        else
            Main.Position = UDim2.fromOffset(dragPosition.X+delta.X+Main.AbsoluteSize.X/2, dragPosition.Y+delta.Y+Main.AbsoluteSize.Y/2)
        end
    end
end)
UserInputService.InputEnded:Connect(function(input) if pointer(input) then dragTarget = nil; resizeTarget = nil end end)

local opened, animating = true, false
local function toggleMenu()
    if animating then return end
    animating = true
    if opened then
        opened = false
        local pos, size = Launcher.AbsolutePosition, Launcher.AbsoluteSize
        local target = UDim2.fromOffset(pos.X+size.X/2, pos.Y+size.Y/2)
        local tween = TweenService:Create(Main, TweenInfo.new(0.4, Enum.EasingStyle.Quart, Enum.EasingDirection.In), {Position = target, Size = UDim2.fromOffset(size.X,size.Y)})
        tween:Play(); tween.Completed:Connect(function() Main.Visible = false; Launcher.Visible = true; Main.Position = UDim2.fromScale(0.5,0.5); Main.Size = UDim2.fromOffset(MenuSize.X,MenuSize.Y); animating = false end)
    else
        opened = true
        local pos, size = Launcher.AbsolutePosition, Launcher.AbsoluteSize
        Launcher.Visible = false; Main.Visible = true
        Main.Position = UDim2.fromOffset(pos.X+size.X/2,pos.Y+size.Y/2); Main.Size = UDim2.fromOffset(size.X,size.Y)
        local tween = TweenService:Create(Main, TweenInfo.new(0.45, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {Position = UDim2.fromScale(0.5,0.5), Size = UDim2.fromOffset(MenuSize.X,MenuSize.Y)})
        tween:Play(); tween.Completed:Connect(function() animating = false end)
    end
end
Launcher.Activated:Connect(toggleMenu)
Close.Activated:Connect(toggleMenu)
